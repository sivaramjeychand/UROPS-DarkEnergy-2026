fig, axes = plt.subplots(2, 1, figsize=(10, 10), sharex=True, gridspec_kw={'hspace': 0.1})

# Original Residuals
axes[0].errorbar(df['z'], df['HR'], yerr=df['e_HR'], fmt='o', alpha=0.6, label='Original', color='blue')
axes[0].axhline(0, color='k', linestyle='--', alpha=0.5)
axes[0].set_ylabel('Hubble Residual (mag)')
axes[0].text(0.05, 0.9, f"RMS = {np.std(df['HR']):.4f} mag", transform=axes[0].transAxes)
axes[0].set_title('Original Hubble Residuals')
axes[0].grid(True, linestyle=':', alpha=0.5)

# Corrected Residuals
axes[1].errorbar(df['z'], df['HR_corr'], yerr=df['e_HR'], fmt='o', alpha=0.6, label='Corrected', color='green')
axes[1].axhline(0, color='k', linestyle='--', alpha=0.5)
axes[1].set_ylabel('Corrected Residual (mag)')
axes[1].set_xlabel('Redshift (z)')
axes[1].text(0.05, 0.9, f"RMS = {np.std(df['HR_corr']):.4f} mag", transform=axes[1].transAxes)
axes[1].set_title('Age-Corrected Hubble Residuals')
axes[1].grid(True, linestyle=':', alpha=0.5)

plt.savefig('fig3_replication.png')
plt.show()

rms_old = np.std(df['HR'])
rms_new = np.std(df['HR_corr'])
improvement = (rms_old - rms_new) / rms_old * 100

print(f"RMS Improvement: {improvement:.2f}%")

# Dispersion in bins (optional check)
bins = np.linspace(df['z'].min(), df['z'].max(), 5)
df['z_bin'] = pd.cut(df['z'], bins)
binned_rms = df.groupby('z_bin', observed=False)[['HR', 'HR_corr']].std()
print("\nBinned RMS:\n", binned_rms)
